        LineRenderer MakeLoop(string name, Material mat, int n, bool loop)
        {
            var g = new GameObject(name); g.transform.SetParent(root.transform, false); var l = g.AddComponent<LineRenderer>(); l.useWorldSpace = true; l.positionCount = n; l.loop = loop; l.sharedMaterial = mat; l.shadowCastingMode = ShadowCastingMode.Off; l.enabled = false; return l;
        }
        Transform GlowQuad(string name, Material mat)
        {
            var q = GameObject.CreatePrimitive(PrimitiveType.Quad); UnityEngine.Object.DestroyImmediate(q.GetComponent<Collider>()); q.name = name; q.transform.SetParent(root.transform, false); q.GetComponent<Renderer>().sharedMaterial = mat; q.GetComponent<Renderer>().shadowCastingMode = ShadowCastingMode.Off; q.SetActive(false); return q.transform;
        }
        void SetLoop(LineRenderer l, Vector3 c, float r, Quaternion q) { int n = l.positionCount; for (int k = 0; k < n; k++) { float a = k * Mathf.PI * 2f / n; l.SetPosition(k, W(c + q * new Vector3(Mathf.Cos(a) * r, 0f, Mathf.Sin(a) * r))); } }
        void Face(Transform q) { var cam = Camera.main; if (cam != null) q.rotation = Quaternion.LookRotation(q.position - cam.transform.position, Vector3.up); }
        LineRenderer[] halos = new LineRenderer[3], baseRings = new LineRenderer[6], towerRings = new LineRenderer[3]; LineRenderer beam; Transform[] tipGlow = new Transform[5], impact = new Transform[6], alarm = new Transform[4], rocks = new Transform[28];
        Vector3[] impactPts = new Vector3[6], rockBase = new Vector3[28]; float[] rockH = new float[28]; int impactN; Material tipMat, impactMat, alarmMat, surfMat; Light alarmLight, officeFlash; GameObject discG; Mesh discMesh; float accRain, accDust2;
        void BuildExtras()
        {
            var soft = kit.Glow(Color.white, lineTex, 3001);
            for (int i = 0; i < halos.Length; i++) halos[i] = MakeLoop("OrbHalo" + i, soft, 72, true);
            for (int i = 0; i < baseRings.Length; i++) baseRings[i] = MakeLoop("BaseRing" + i, soft, 48, true);
            for (int i = 0; i < towerRings.Length; i++) towerRings[i] = MakeLoop("TowerRing" + i, soft, 48, true);
            beam = MakeLoop("OrbBeam", soft, 2, false);
            tipMat = kit.Glow(new Color(1f, 0.75f, 0.9f, 0.9f), glowTex, 3003); impactMat = kit.Glow(new Color(1f, 0.7f, 0.95f, 0.85f), glowTex, 3003); alarmMat = kit.Glow(new Color(1f, 0.15f, 0.1f, 0.9f), glowTex, 3003);
            for (int i = 0; i < tipGlow.Length; i++) tipGlow[i] = GlowQuad("TipGlow" + i, tipMat); for (int i = 0; i < impact.Length; i++) impact[i] = GlowQuad("Impact" + i, impactMat); for (int i = 0; i < alarm.Length; i++) alarm[i] = GlowQuad("Alarm" + i, alarmMat);
            alarmLight = new GameObject("AlarmLight").AddComponent<Light>(); alarmLight.transform.SetParent(root.transform, false); alarmLight.type = LightType.Point; alarmLight.color = new Color(1f, 0.15f, 0.1f); alarmLight.range = 80f * sc + 20f; alarmLight.intensity = 0f; alarmLight.shadows = LightShadows.None; alarmLight.transform.localPosition = new Vector3(0f, gy + 22f, 0f);
            var rm = kit.Lit(new Color(0.30f, 0.27f, 0.30f), null, 0.1f);
            for (int i = 0; i < rocks.Length; i++)
            {
                float ang = WorldKit.Hash(i, 1, 5) * 6.2832f, rad = 14f + WorldKit.Hash(i, 2, 5) * 46f, s = 1.4f + WorldKit.Hash(i, 3, 5) * 3.6f; rockH[i] = WorldKit.Hash(i, 4, 5); rockBase[i] = new Vector3(Mathf.Cos(ang) * rad, gy + 0.5f, Mathf.Sin(ang) * rad);
                var o = kit.Obj("FloatRock" + i, root.transform, kit.Blob(1f, 0.7f, 0.9f, 11 + i, 0.3f, 8, 12), rm, rockBase[i], Quaternion.identity, Vector3.one * s, false).transform; o.gameObject.SetActive(false); rocks[i] = o;
            }
            // the flooded island: a flat disc inside the water wall (the sea outside it is the annulus)
            var v = new List<Vector3>(); var uv = new List<Vector2>(); var tr = new List<int>(); v.Add(Vector3.zero); uv.Add(Vector2.zero); int seg = 48;
            for (int j = 0; j <= seg; j++) { float a = j * Mathf.PI * 2f / seg; v.Add(new Vector3(Mathf.Cos(a), 0f, Mathf.Sin(a))); uv.Add(new Vector2(Mathf.Cos(a) * 2f, Mathf.Sin(a) * 2f)); }
            for (int j = 1; j <= seg; j++) { tr.Add(0); tr.Add(j + 1); tr.Add(j); }
            discMesh = new Mesh(); discMesh.SetVertices(v); discMesh.SetUVs(0, uv); discMesh.SetTriangles(tr, 0); discMesh.RecalculateBounds(); kit.Keep(discMesh);
            surfMat = kit.Glow(Color.white, waterMat.mainTexture as Texture2D, 2994); discG = new GameObject("WaterSurface"); discG.transform.SetParent(root.transform, false); discG.AddComponent<MeshFilter>().sharedMesh = discMesh; var mr = discG.AddComponent<MeshRenderer>(); mr.sharedMaterial = surfMat; mr.shadowCastingMode = ShadowCastingMode.Off; discG.SetActive(false);
        }
        float WaterLevel(float t) { return (0.15f + 1.45f * Sm(340f, 398f, t)) / sc; }              // the flood is 1.6 m deep (real metres) at the end: deep enough to swim in
        void SwimStep(float t)
        {
            if (paused) return; bool want = false; var hp = Head();
            if (hp.HasValue && WaterA(t) > 0.3f) { float sy = W(new Vector3(0f, gy + WaterLevel(t), 0f)).y; var c = W(Vector3.zero); float dx = hp.Value.x - c.x, dz = hp.Value.z - c.z; want = hp.Value.y < sy && Mathf.Sqrt(dx * dx + dz * dz) < WaterR(t) * sc * 1.2f; }
            AstronomicalHazards.Swim(want, W(Vector3.zero));
        }
        void Extras(float t, float dt, float pulse)
        {
            bool orb = OrbOn(t); float act = orb ? Mathf.Clamp01(OrbBolts(t) / 5f) : 0f; Vector3 oc = orb ? OrbCentre(t) : Vector3.zero; float e = TowerE(t);
            for (int i = 0; i < halos.Length; i++)
            {
                var h = halos[i]; bool on = orb && Sm(111f, 116f, t) > 0.05f; if (h.enabled != on) h.enabled = on; if (!on) continue;
                var q = Quaternion.Euler(70f * Mathf.Sin(t * 0.35f + i * 2.1f), t * (22f + 14f * i) * (i % 2 == 0 ? 1f : -1f), 35f * Mathf.Cos(t * 0.3f + i)); SetLoop(h, oc, (15f + 6f * i) * (1f + 0.05f * pulse), q);
                var c = i == 0 ? new Color(1f, 0.6f, 0.85f) : i == 1 ? new Color(0.6f, 0.85f, 1f) : Color.white; h.startColor = h.endColor = new Color(c.r, c.g, c.b, (0.25f + 0.55f * act) * (1f - Sm(294f, 298f, t))); h.widthMultiplier = (0.5f + 0.5f * pulse) * sc;
            }
            bool bon = orb && act > 0.01f; if (beam.enabled != bon) beam.enabled = bon; if (bon) { beam.SetPosition(0, W(oc)); beam.SetPosition(1, W(oc + Vector3.up * 450f)); beam.widthMultiplier = (5f + 4f * act + 3f * pulse) * sc; beam.startColor = beam.endColor = new Color(1f, 0.8f, 0.95f, 0.22f + 0.2f * act); }
            for (int i = 0; i < tipGlow.Length; i++)
            {
                bool on = PylonE(i, t) > 0.5f; if (tipGlow[i].gameObject.activeSelf != on) tipGlow[i].gameObject.SetActive(on); if (!on) continue;
                bool live = Array.IndexOf(PYL_ORDER, i) < OrbBolts(t); tipGlow[i].position = W(PylonTip(i, t)); tipGlow[i].localScale = Vector3.one * ((live ? 16f + 8f * pulse : 6f) * sc); Face(tipGlow[i]);
            }
            for (int i = 0; i < baseRings.Length; i++)
            {
                var r = baseRings[i]; bool up = i < 5 ? PylonE(i, t) > 0.5f : DeviceY(t) > -8f; bool on = up && act > 0.01f; if (r.enabled != on) r.enabled = on; if (!on) continue;
                SetLoop(r, (i < 5 ? PylonBase(i) : new Vector3(0f, gy, 0f)) + Vector3.up * 0.4f, (i < 5 ? 6f : 13f) * (1f + 0.15f * pulse), Quaternion.identity); var c = i % 2 == 0 ? new Color(1f, 0.6f, 0.85f) : new Color(0.6f, 0.85f, 1f); r.startColor = r.endColor = new Color(c.r, c.g, c.b, 0.8f * act); r.widthMultiplier = 1.1f * sc;
            }
            float tower = e * 72f * TOWER_K; for (int k = 0; k < towerRings.Length; k++)
            {
                var r = towerRings[k]; bool on = orb && e > 0.5f && act > 0.01f; if (r.enabled != on) r.enabled = on; if (!on) continue;
                float f = ((t * 0.5f + k / 3f) % 1f); float y = gy + 3f + tower * (1f - f); SetLoop(r, new Vector3(0f, y, 0f), 4.2f * TOWER_K + 0.6f, Quaternion.identity); r.startColor = r.endColor = new Color(1f, 0.65f, 0.9f, 0.85f * act * Mathf.Sin(f * Mathf.PI)); r.widthMultiplier = 0.9f * sc;
            }
            float lift = Sm(195f, 215f, t) * (1f - Sm(256f, 262f, t)); bool ron = lift > 0.001f;
            for (int i = 0; i < rocks.Length; i++)
            {
                if (rocks[i].gameObject.activeSelf != ron) rocks[i].gameObject.SetActive(ron); if (!ron) continue;
                rocks[i].localPosition = rockBase[i] + Vector3.up * (lift * (7f + 18f * rockH[i]) + 1.2f * lift * Mathf.Sin(t * 0.8f + i)); rocks[i].localRotation = Quaternion.Euler(t * 9f * (rockH[i] - 0.4f), t * 14f * rockH[i], t * 6f * (0.5f - rockH[i]));
            }
            for (int i = 0; i < impact.Length; i++)
            {
                bool on = i < impactN; if (impact[i].gameObject.activeSelf != on) impact[i].gameObject.SetActive(on); if (!on) continue; impact[i].position = W(impactPts[i]); impact[i].localScale = Vector3.one * ((26f + 12f * pulse) * sc); Face(impact[i]);
            }
            bool al = t >= 30f && t < 71f && agencyT != null; float fl = al ? Mathf.Pow(Mathf.Max(0f, Mathf.Sin(t * 6f)), 2f) : 0f; alarmLight.intensity = Lit * 4f * fl;
            for (int i = 0; i < alarm.Length; i++) { if (alarm[i].gameObject.activeSelf != al) alarm[i].gameObject.SetActive(al); if (!al) continue; alarm[i].localPosition = new Vector3((i & 1) == 0 ? -12f : 12f, gy + 20.5f, (i & 2) == 0 ? -7f : 7f); alarm[i].localScale = Vector3.one * ((6f + 10f * fl) * sc); Face(alarm[i]); }
            if (paused || dt <= 0f) return;
            if (t >= 55f && t < 71f) for (int k = Take(ref accDust2, 18f * Parts * Sm(55f, 71f, t), dt); k > 0; k--) Emit(smoke, new Vector3(Rnd(-22f, 22f), gy + 1f, Rnd(-18f, 18f)), new Vector3(Rnd(-1f, 1f), Rnd(3f, 7f), Rnd(-1f, 1f)), new Color(0.4f, 0.38f, 0.36f, 0.5f), Rnd(5f, 9f), Rnd(5f, 7f));
            if (t >= 330f && t < 400f) for (int k = Take(ref accRain, 260f * Parts, dt); k > 0; k--) Emit(fx, new Vector3(Rnd(-70f, 70f), gy + Rnd(25f, 45f), Rnd(-70f, 70f)), new Vector3(Rnd(-3f, 3f), -Rnd(55f, 70f), Rnd(-3f, 3f)), new Color(0.75f, 0.9f, 1f, 0.5f), 0.35f, 0.55f);
        }
